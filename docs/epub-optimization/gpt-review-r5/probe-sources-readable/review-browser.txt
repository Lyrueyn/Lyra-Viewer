async function make(name,bodies){const z=new JSZip();
z.file('mimetype','application/epub+zip');
z.file('META-INF/container.xml','<container xmlns="urn:oasis:names:tc:opendocument:xmlns:container"><rootfiles><rootfile full-path="OPS/book.opf"/></rootfiles></container>');
z.file('OPS/book.opf',`<package xmlns="http://www.idpf.org/2007/opf" version="3.0"><metadata xmlns:dc="http://purl.org/dc/elements/1.1/"><dc:title>${name}</dc:title></metadata><manifest><item id="nav" href="nav.xhtml" media-type="application/xhtml+xml" properties="nav"/>${bodies.map((_,i)=>`<item id="c${i}" href="c${i}.xhtml" media-type="application/xhtml+xml"/>`).join('')}</manifest><spine>${bodies.map((_,i)=>`<itemref idref="c${i}"/>`).join('')}</spine></package>`);
z.file('OPS/nav.xhtml',`<html xmlns="http://www.w3.org/1999/xhtml" xmlns:epub="http://www.idpf.org/2007/ops"><body><nav epub:type="toc"><ol>${bodies.map((_,i)=>`<li><a href="c${i}.xhtml">CH-${i}</a></li>`).join('')}</ol></nav></body></html>`);
bodies.forEach((b,i)=>z.file(`OPS/c${i}.xhtml`,`<html xmlns="http://www.w3.org/1999/xhtml"><body><h1>${name}-CH-${i}</h1>${b}</body></html>`));
return z.generateAsync({type:'nodebuffer',compression:'DEFLATE'});
}
