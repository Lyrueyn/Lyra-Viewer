const fs=require('node:fs'),path=require('node:path'),http=require('node:http');
const{chromium}=require(process.env.PLAYWRIGHT_MODULE);
const root=path.join(__dirname,'candidate'),fixtures=path.join(__dirname,'advanced-evidence'),out=path.join(__dirname,'storage-evidence');
fs.mkdirSync(out,{recursive:true});
const results=[];
(async()=>{const server=http.createServer((req,res)=>{const p=new URL(req.url,'http://localhost').pathname,file=p.startsWith('/fixtures/')?path.join(fixtures,path.basename(p)):path.join(root,p==='/'?'index.html':p);
if(!fs.existsSync(file)||fs.statSync(file).isDirectory()){res.writeHead(404);
res.end();
return;
}res.writeHead(200,{'Content-Type':file.endsWith('.js')?'application/javascript':file.endsWith('.css')?'text/css':'text/html;
 charset=utf-8'});
fs.createReadStream(file).pipe(res);
});
await new Promise(r=>server.listen(0,'127.0.0.1',r));
const browser=await chromium.launch({headless:true,executablePath:process.env.LYRA_BROWSER_PATH});
let ctx,page;
async function open(){if(ctx)await ctx.close();
ctx=await browser.newContext({viewport:{width:1400,height:900}});
await ctx.addInitScript(()=>window.__LYRA_TEST_MODE__=true);
page=await ctx.newPage();
await page.goto('http://127.0.0.1:'+server.address().port+'/index.html?book=fixtures/normal.epub');
await page.waitForFunction(()=>__lyraDebug?.state.ready);
}async function probe(id,f){try{const r={id,...await f()};
results.push(r);
console.log(JSON.stringify(r));
}catch(e){results.push({id,probeError:e.message});
}fs.writeFileSync(path.join(out,'results.json'),JSON.stringify(results,null,2));
}try{for(const kind of ['position','preferences','bookmarks'])await probe('B13_CORRUPT_LOCAL_'+kind.toUpperCase(),async()=>{await open();
const key=await page.evaluate(kind=>{const id=__lyraDebug.state.bookId,k=kind==='position'?'lyra_epub_state_'+id:kind==='preferences'?'lyra_epub_prefs':'lyra_epub_bookmarks_'+id;
localStorage.setItem(k,'{invalid-json');
return k;
},kind);
await page.reload();
await page.waitForFunction(()=>__lyraDebug?.state.ready||document.getElementById('errorBox').classList.contains('show'));
const r=await page.evaluate(key=>({ready:__lyraDebug.state.ready,error:document.getElementById('errorText').textContent,raw:localStorage.getItem(key)}),key);
return{...r,defect:!r.ready,preserved:r.raw==='{invalid-json'};
});
await probe('B14_FAILED_SAVE_SWITCH_LOSES_PENDING_POSITION',async()=>{await open();
const old=await page.evaluate(()=>__lyraDebug.state.bookId);
await page.evaluate(()=>{const c=__lyraDebug.state.client,orig=c.savePosition.bind(c);
window.__writes=[];
c.savePosition=async(...args)=>{__writes.push(args[0]);
if(!window.__storageRecovered)throw Object.assign(new Error('Controlled temporary storage failure'),{code:'LOCAL_WRITE_FAILED'});
return orig(...args);
};
});
await page.locator('#pageInput').fill('7');
await page.locator('#pageInput').press('Enter');
await page.waitForTimeout(120);
await page.locator('#file').setInputFiles(path.join(fixtures,'fresh.epub'));
await page.waitForFunction(()=>__lyraDebug.state.ready&&__lyraDebug.state.book.title==='FRESH');
await page.evaluate(()=>window.__storageRecovered=true);
await page.waitForTimeout(550);
const r=await page.evaluate(old=>({book:__lyraDebug.state.book.title,modified:__lyraDebug.state.modified,oldSaved:localStorage.getItem('lyra_epub_state_'+old),writes:__writes,status:document.getElementById('saveStateText').textContent}),old);
return{...r,defect:r.oldSaved===null&&!r.modified,note:'Controlled temporary storage failure;
 verifies whether the old-book unsaved snapshot survives switching and later storage recovery.'};
});
}finally{if(ctx)await ctx.close();
await browser.close();
await new Promise(r=>server.close(r));
fs.writeFileSync(path.join(out,'results.json'),JSON.stringify(results,null,2));
}})().catch(e=>{console.error(e);
process.exitCode=1;
});
