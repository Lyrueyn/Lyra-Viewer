async function make(title,bodies){const z=new JSZip();
z.file('mimetype','application/epub+zip');
z.file('META-INF/container.xml','<container><rootfiles><rootfile full-path="OPS/book.opf"/></rootfiles></container>');
z.file('OPS/book.opf',`<package xmlns="http://www.idpf.org/2007/opf" version="3.0"><metadata xmlns:dc="http://purl.org/dc/elements/1.1/"><dc:title>${title}</dc:title></metadata><manifest><item id="nav" href="nav.xhtml" media-type="application/xhtml+xml" properties="nav"/>${bodies.map((_,i)=>`<item id="c${i}" href="c${i}.xhtml" media-type="application/xhtml+xml"/>`).join('')}</manifest><spine>${bodies.map((_,i)=>`<itemref idref="c${i}"/>`).join('')}</spine></package>`);
z.file('OPS/nav.xhtml',`<html xmlns:epub="http://www.idpf.org/2007/ops"><body><nav epub:type="toc"><ol>${bodies.map((_,i)=>`<li><a href="c${i}.xhtml">CH-${i}</a></li>`).join('')}</ol></nav></body></html>`);
bodies.forEach((body,i)=>z.file(`OPS/c${i}.xhtml`,`<html><body><h1>${title}-${i}</h1>${body}</body></html>`));
return z.generateAsync({type:'nodebuffer',compression:'DEFLATE'});
}

res.writeHead(b?200:404);
res.end(b);

res.end();

}res.writeHead(200,{'Content-Type':file.endsWith('.js')?'application/javascript':file.endsWith('.css')?'text/css':'text/html;
 charset=utf-8'});
fs.createReadStream(file).pipe(res);
});
