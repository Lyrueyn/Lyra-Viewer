const fs=require('node:fs'),path=require('node:path'),http=require('node:http');
const{chromium}=require(process.env.PLAYWRIGHT_MODULE);
const root=path.join(__dirname,'candidate'),out=path.join(__dirname,'anchor-evidence');
fs.mkdirSync(out,{recursive:true});

(async()=>{const server=http.createServer((req,res)=>{const p=new URL(req.url,'http://localhost').pathname,file=p==='/test.epub'?path.join(__dirname,'boundary-evidence','highlight.epub'):path.join(root,p==='/'?'index.html':p);
if(!fs.existsSync(file)){res.writeHead(404);
res.end();
return;
}res.writeHead(200,{'Content-Type':file.endsWith('.js')?'application/javascript':file.endsWith('.css')?'text/css':'text/html;
 charset=utf-8'});
fs.createReadStream(file).pipe(res);
});
await new Promise(r=>server.listen(0,'127.0.0.1',r));
const browser=await chromium.launch({headless:true,executablePath:process.env.LYRA_BROWSER_PATH});
try{const page=await browser.newPage({viewport:{width:1400,height:900}});
await page.addInitScript(()=>{window.__LYRA_TEST_MODE__=true;
});
await page.goto('http://127.0.0.1:'+server.address().port+'/index.html?book=test.epub');
await page.waitForFunction(()=>__lyraDebug?.state.ready);
await page.evaluate(()=>{const p=__lyraDebug.paginator,n=p.content.querySelector('#long').firstChild,start=n.data.indexOf('UNIQUE_TARGET'),r=p.doc.createRange();
r.setStart(n,start);
r.setEnd(n,start+'UNIQUE_TARGET'.length);
p.highlightRanges([r]);
});
await page.locator('#pageInput').fill('8');
await page.locator('#pageInput').press('Enter');
await page.waitForTimeout(500);
const before=await page.evaluate(()=>({page:__lyraDebug.paginator.pageIndex,anchor:__lyraDebug.paginator.captureAnchor(),highlights:__lyraDebug.paginator.content.querySelectorAll('mark').length}));
await page.reload();
await page.waitForFunction(()=>__lyraDebug?.state.ready);
const after=await page.evaluate(()=>({page:__lyraDebug.paginator.pageIndex,anchor:__lyraDebug.paginator.captureAnchor()}));
const result={id:'R18_HIGHLIGHT_ANCHOR_DIRECT',before,after,defect:before.page!==after.page,note:'Invoke candidate highlightRanges with a valid range, then use actual page input and reload. This isolates anchor persistence from the separately broken search locator.'};
fs.writeFileSync(path.join(out,'anchor-result.json'),JSON.stringify(result,null,2));
console.log(JSON.stringify(result));
}finally{await browser.close();
await new Promise(r=>server.close(r));
}})().catch(e=>{console.error(e);
process.exitCode=1;
});
